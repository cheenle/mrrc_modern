# W103D 盒子 DDNS（w103d.vlsc.net AAAA）实现计划

> **面向 AI 代理的工作者：** 必需子技能：使用 superpowers:subagent-driven-development（推荐）或 superpowers:executing-plans 逐任务实现此计划。步骤使用复选框（`- [ ]`）语法来跟踪进度。

**目标：** 在 W103D 盒子（192.168.1.217）上跑一个零依赖的 DDNS 自更新器，把 `w103d.vlsc.net` 的 AAAA 记录追着盒子当前出网 IPv6 走。

**架构：** 单 bash 脚本（`curl`+`openssl`+`jq`，全部已装）负责"取地址 → Describe → 比对 → 增/改/不动"；配置在 600 权限的 `ddns.env`；systemd oneshot + timer（开机 90 秒 + 每 5 分钟）驱动。

**技术栈：** bash、AliDNS OpenAPI（HMAC-SHA1 签名，python3 stdlib 实现编码/签名）、jq（JSON 解析）、systemd timer、dig（验收）。

**设计依据：** `docs/superpowers/specs/2026-10-11-w103d-ddns-design.md`

- [ ] **步骤 1：写 `w103d-ddns.sh`**

```bash
#!/usr/bin/env bash
# w103d-ddns —— 把 w103d.vlsc.net 的 AAAA 记录追着本机出网 IPv6 走
#
# 设计：mrrc_modern/docs/superpowers/specs/2026-10-11-w103d-ddns-design.md
# 计划：mrrc_modern/docs/superpowers/plans/2026-10-11-w103d-ddns.md
#
# 退出码：0=成功/无变化  1=配置或取地址失败  2=记录数异常  3=AliDNS API 失败  64=参数错误
set -euo pipefail

ENV_FILE="${W103D_ENV_FILE:-/opt/w103d-ddns/ddns.env}"
API="https://alidns.aliyuncs.com/"
DRY_RUN=0

log() { printf 'w103d-ddns: %s\n' "$*"; }
die() { printf 'w103d-ddns: ERROR: %s\n' "$*" >&2; exit "${2:-1}"; }

case "${1:-}" in
  --dry-run) DRY_RUN=1 ;;
  "")        ;;
  -h|--help) echo "用法: w103d-ddns [--dry-run]"; exit 0 ;;
  *)         echo "用法: w103d-ddns [--dry-run]" >&2; exit 64 ;;
esac

[ -r "$ENV_FILE" ] || die "读不到配置文件 $ENV_FILE" 1
# shellcheck source=/dev/null
. "$ENV_FILE"

: "${ALI_ACCESS_KEY_ID:?ALI_ACCESS_KEY_ID 未设置（见 $ENV_FILE）}"
: "${ALI_ACCESS_SECRET:?ALI_ACCESS_SECRET 未设置（见 $ENV_FILE）}"
DOMAIN="${W103D_DOMAIN:-vlsc.net}"
RECORD="${W103D_RECORD:-w103d}"
TTL="${W103D_TTL:-600}"
FQDN="$RECORD.$DOMAIN"

# 同一时刻只允许一个实例（timer 与手动运行不打架）
exec 9>/run/w103d-ddns.lock
flock -n 9 || { log "上一轮尚未结束，跳过本轮"; exit 0; }

# ---------- 1. 取地址：由内核裁定“出网用哪个源地址” ----------
is_global_v6() { case "$1" in [23][0-9a-fA-F][0-9a-fA-F][0-9a-fA-F]:*) return 0 ;; *) return 1 ;; esac; }

pick_address() {
  local src dev
  src="$(ip -6 route get 2400:3200::1 2>/dev/null \
        | awk '{for(i=1;i<=NF;i++) if($i=="src"){print $(i+1); exit}}')" || true
  if [ -n "${src:-}" ] && is_global_v6 "$src"; then printf '%s\n' "$src"; return 0; fi
  # 回退：默认路由所在接口上的第一个全球地址
  dev="$(ip -6 route show default 2>/dev/null \
        | awk '{for(i=1;i<=NF;i++) if($i=="dev"){print $(i+1); exit}}')" || true
  [ -n "${dev:-}" ] || return 1
  ip -6 addr show dev "$dev" scope global 2>/dev/null \
    | awk '/inet6/ && !/temporary|deprecated/ {sub(/\/.*/,"",$2); print $2}' \
    | while read -r a; do is_global_v6 "$a" && { printf '%s\n' "$a"; break; }; done
}

if [ -n "${W103D_FORCE_ADDRESS:-}" ]; then
  ADDR="$W103D_FORCE_ADDRESS"
  log "W103D_FORCE_ADDRESS=$ADDR（测试钩子）"
else
  ADDR="$(pick_address || true)"
fi
[ -n "${ADDR:-}" ] || die "找不到全局 IPv6 地址（未联网，或只有 ULA/链路本地）" 1
is_global_v6 "$ADDR" || die "地址 $ADDR 不在 2000::/3，拒绝写进 DNS" 1
log "本机出网地址：$ADDR"

# ---------- 2. AliDNS 签名（RFC3986 + HMAC-SHA1） ----------
# 编码/签名全在 python3 stdlib 里做。实测 jq 的 @uri 会把 ! ' ( ) * 留在外面，
# 不符合 RFC3986（阿里云要求这些字符也百分号编码），故不用 jq 做编码。
# jq 只用于后面的 JSON 解析。python3 为 Armbian 自带。
SIGNER='
import base64, hashlib, hmac, sys, urllib.parse

secret = sys.argv[1]
pairs = []
for line in sys.stdin.read().splitlines():
    k, v = line.split("=", 1)
    pairs.append((urllib.parse.quote(k, safe=""), urllib.parse.quote(v, safe="")))
canon = "&".join("%s=%s" % (k, v) for k, v in sorted(pairs))
sts = "GET&%2F&" + urllib.parse.quote(canon, safe="")
sig = base64.b64encode(
    hmac.new((secret + "&").encode(), sts.encode(), hashlib.sha1).digest()
).decode()
print(canon + "&Signature=" + urllib.parse.quote(sig, safe=""))
'

api_call() {  # api_call Key=Value ...
  local qs
  qs="$(
    { printf 'AccessKeyId=%s\nFormat=json\nSignatureMethod=HMAC-SHA1\nSignatureNonce=%s\nSignatureVersion=1.0\nTimestamp=%s\nVersion=2015-01-09\n' \
        "$ALI_ACCESS_KEY_ID" "$(cat /proc/sys/kernel/random/uuid)" "$(date -u '+%Y-%m-%dT%H:%M:%SZ')"
      printf '%s\n' "$@"
    } | python3 -c "$SIGNER" "$ALI_ACCESS_SECRET"
  )" || return 1
  [ -n "$qs" ] || return 1
  curl -sS --max-time 20 --retry 2 --retry-delay 1 "$API?$qs"
}

RESP=""
api_json() {  # 同 api_call，但校验 JSON / API 错误；结果放 $RESP
  local code rc=0
  RESP="$(api_call "$@")" || rc=$?
  [ "$rc" -eq 0 ] || die "调用阿里云 API 失败（网络/超时，curl 退出码 $rc）" 3
  printf '%s' "$RESP" | jq -e . >/dev/null 2>&1 || die "阿里云返回的不是 JSON：$(printf '%.200s' "$RESP")" 3
  code="$(printf '%s' "$RESP" | jq -r '.Code // empty')"
  [ -z "$code" ] || die "阿里云 API 报错 [$code] $(printf '%s' "$RESP" | jq -r '.Message // ""')" 3
}

# ---------- 3. 查记录 → 比对 → 增/改/不动 ----------
api_json "Action=DescribeSubDomainRecords" "SubDomain=$FQDN" "Type=AAAA"
total="$(printf '%s' "$RESP" | jq -r '.TotalCount')"
current="$(printf '%s' "$RESP" | jq -r '.DomainRecords.Record[0].Value // empty')"
record_id="$(printf '%s' "$RESP" | jq -r '.DomainRecords.Record[0].RecordId // empty')"

# IPv6 文本形式归一（压缩/大写差异不误判）；python3 缺失时退化为小写比较
norm6() {
  python3 -c 'import ipaddress,sys;print(ipaddress.IPv6Address(sys.argv[1]))' "$1" 2>/dev/null \
    || printf '%s' "$1" | tr 'A-F' 'a-f'
}

case "$total" in
  0)
    if [ "$DRY_RUN" = 1 ]; then log "[dry-run] 将新建 AAAA $FQDN → $ADDR"; exit 0; fi
    api_json "Action=AddDomainRecord" "DomainName=$DOMAIN" "RR=$RECORD" "Type=AAAA" "Value=$ADDR" "TTL=$TTL"
    log "已新建 AAAA：$FQDN → $ADDR"
    ;;
  1)
    if [ "$(norm6 "$current")" = "$(norm6 "$ADDR")" ]; then
      log "无变化：$FQDN = $ADDR"
      exit 0
    fi
    if [ "$DRY_RUN" = 1 ]; then log "[dry-run] 将更新 AAAA $FQDN：$current → $ADDR"; exit 0; fi
    api_json "Action=UpdateDomainRecord" "RecordId=$record_id" "RR=$RECORD" "Type=AAAA" "Value=$ADDR" "TTL=$TTL"
    log "已更新 AAAA：$FQDN  $current → $ADDR"
    ;;
  *)
    die "$FQDN 有 $total 条 AAAA 记录，数量异常，请人工核查后再跑" 2
    ;;
esac
```

把 `w103d.vlsc.net` 的 AAAA 记录追着本机出网 IPv6 走。每 5 分钟由 systemd timer 跑一次。

    0=成功/无变化  1=配置或取地址失败  2=AAAA 记录数异常  3=AliDNS API 失败  64=参数错误

- 周期：改 `/etc/systemd/system/w103d-ddns.timer` 的 `OnUnitActiveSec` 后
  `systemctl daemon-reload && systemctl restart w103d-ddns.timer`
- 密钥/RR/TTL：改 `/opt/w103d-ddns/ddns.env`（600），下一轮生效

## 故障定位

- `找不到全局 IPv6 地址` → 盒子没连上带 v6 的网络：`ip -6 addr` / `ip -6 route`
- `阿里云 API 报错 [SignatureDoesNotMatch]` → `ddns.env` 里密钥被改动
- `记录数异常` → 域名控制台里 `w103d` 有重复 AAAA，人工删到只剩一条
- 抢锁跳过（`上一轮尚未结束`）→ 上一轮卡住了：`journalctl` 看原因，必要时手动再跑

## 卸载

    systemctl disable --now w103d-ddns.timer
    rm -rf /opt/w103d-ddns /etc/systemd/system/w103d-ddns.{service,timer}
    systemctl daemon-reload
    # DNS 记录需自行到阿里云控制台/API 删除

## 安全警示
