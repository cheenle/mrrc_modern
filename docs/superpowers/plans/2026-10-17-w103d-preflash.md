# W103D 发货版（预置）镜像 实现计划

> **面向 AI 代理的工作者：** 必需子技能：使用 superpowers:subagent-driven-development（推荐）或 superpowers:executing-plans 逐任务实现此计划。步骤使用复选框（`- [ ]`）语法跟踪进度。
> 唯一依据：`docs/superpowers/specs/2026-10-17-w103d-preflash-design.md`。

**目标：** 让"卖家代刷一批 W103D、快递给用户"这条路零初始化：镜像出厂就带固定口令（`root`=`abcd1234`、`mrrc`=`abcd!1234`、Web=`abcd!1234`）与系统层预置（`Asia/Shanghai` + `zh_CN.UTF-8`）；Wi-Fi 与公网入口都在盒子设置页 `/settings` 里配（Wi-Fi 支持手输 SSID 与明文查看密码，公网入口支持填呼号向 portal 申请）；并把"首启探测"从"跑过一次"改成"跑到定型为止"，于是用户家的电台无论什么时候插上都能被发现。

**架构：** 三层各加一处、互不耦合。①**构建层**：`build-image.sh --preflash`（默认关）把 `MRRC_PREFLASH=1` 经既有的容器→chroot 环境变量通道传进去，共享服务层在最后调用新脚本 `packaging/common/preflash.sh`（写 env 固定口令、`chpasswd`、时区、locale）。②**页面层**：`static/settings.html` 加两块面板，只读/只调既有端点（`/api/box/wifi`、`/api/cloud/{state,apply,refresh}`），服务端零改动。③**首启层**：`linux/first_run.py` 的 `MRRC_PRESEED=1` 从"短路返回"改成"不生成口令、其余照常"，`firstboot_wrapper.py` 只在配置定型时才盖 `firstboot-done`。

**技术栈：** bash（POSIX 兼容到 bash 3.2，仓库有守卫测试）、Python 3.11（stdlib + pyserial；单测用 `unittest`）、原生 ES5 风格 JS（无构建步骤）、Docker 特权容器（构建与产物验证）。

```python
"""出厂预置脚本：真跑它，读它写出来的文件。

规格 D-3 的两条硬要求都能用一次真实执行验掉 —— 固定口令写进 env、以及
`MRRC_PREFLASH` 没置 1 时一个字都不写。脚本是 shell，所以断言只能落在
它写出来的文件上；`--env-only` 模式存在的理由就是这个测试。
"""
from __future__ import annotations

import os
import subprocess
import tempfile
import unittest
from pathlib import Path

REPO = Path(__file__).resolve().parents[1]
SCRIPT = REPO / "packaging" / "common" / "preflash.sh"

BASE_ENV = """\
# MRRC Modern Configuration
MRRC_BAUD_RATE=38400
MRRC_WEB_PORT=8888
MRRC_WEB_HOST=0.0.0.0
MRRC_PTT_MAX_TX_SECONDS=120
MRRC_SSL_CERT=/var/lib/mrrc/certs/server.crt
"""


def run_preflash(env_file: Path, *, flag: str | None = "1", extra: dict | None = None):
    env = {**os.environ, **(extra or {})}
    env.pop("MRRC_PREFLASH", None)
    if flag is not None:
        env["MRRC_PREFLASH"] = flag
    return subprocess.run(["bash", str(SCRIPT), "--env-only", str(env_file)],
                          capture_output=True, text=True, env=env)


def key_count(text: str, key: str) -> int:
    return sum(1 for line in text.splitlines() if line.startswith(key + "="))


class PreflashEnvTests(unittest.TestCase):
    def setUp(self):
        self.tmp = tempfile.TemporaryDirectory()
        self.addCleanup(self.tmp.cleanup)
        self.env_file = Path(self.tmp.name) / "mrrc.env"
        self.env_file.write_text(BASE_ENV, encoding="utf-8")

    def test_it_writes_the_fixed_password_and_clears_the_auto_flag(self):
        result = run_preflash(self.env_file)
        self.assertEqual(result.returncode, 0, result.stderr)
        text = self.env_file.read_text(encoding="utf-8")
        self.assertIn("MRRC_WEB_PASSWORD=abcd!1234", text)
        self.assertEqual(key_count(text, "MRRC_WEB_PASSWORD"), 1)
        self.assertIn("MRRC_AUTO_PASSWORD=\n", text,
                      "AUTO 标志必须显式清空，否则页面会当它从未被设过")

    def test_it_keeps_every_other_key(self):
        run_preflash(self.env_file)
        text = self.env_file.read_text(encoding="utf-8")
        for line in BASE_ENV.splitlines():
            if not line or line.startswith("#"):
                continue
            with self.subTest(line=line):
                self.assertIn(line, text)

    def test_it_does_not_preset_a_radio_model_or_port(self):
        """型号与串口由设备上的探测决定（规格 D-3 / NFR-067）。"""
        run_preflash(self.env_file)
        text = self.env_file.read_text(encoding="utf-8")
        self.assertNotIn("MRRC_RADIO_MODEL=", text)
        self.assertNotIn("MRRC_SERIAL_PORT=", text)

    def test_it_is_idempotent(self):
        run_preflash(self.env_file)
        once = self.env_file.read_bytes()
        run_preflash(self.env_file)
        self.assertEqual(once, self.env_file.read_bytes())

    def test_it_replaces_an_existing_password_instead_of_appending(self):
        self.env_file.write_text(BASE_ENV + "MRRC_WEB_PASSWORD=old-one\n", encoding="utf-8")
        run_preflash(self.env_file)
        text = self.env_file.read_text(encoding="utf-8")
        self.assertEqual(key_count(text, "MRRC_WEB_PASSWORD"), 1)
        self.assertIn("MRRC_WEB_PASSWORD=abcd!1234", text)
        self.assertNotIn("old-one", text)

    def test_the_environment_can_override_the_batch_values(self):
        run_preflash(self.env_file, extra={"MRRC_PREFLASH_WEB_PASSWORD": "batch-two"})
        self.assertIn("MRRC_WEB_PASSWORD=batch-two",
                      self.env_file.read_text(encoding="utf-8"))

    def test_the_shell_metacharacter_in_the_default_password_survives(self):
        """口令里的 `!` 是构建→env→systemd→chpasswd 四层里的一个真风险点。"""
        run_preflash(self.env_file)
        line = [l for l in self.env_file.read_text(encoding="utf-8").splitlines()
                if l.startswith("MRRC_WEB_PASSWORD=")][0]
        self.assertEqual(line, "MRRC_WEB_PASSWORD=abcd!1234")

    def test_it_refuses_to_write_anything_without_the_flag(self):
        before = self.env_file.read_bytes()
        result = run_preflash(self.env_file, flag=None)
        self.assertNotEqual(result.returncode, 0,
                           "开关没置 1 也照写 == 默认镜像被悄悄改了口令")
        self.assertEqual(before, self.env_file.read_bytes())
        self.assertIn("MRRC_PREFLASH", result.stderr + result.stdout)


class BuildWiringTests(unittest.TestCase):
    """构建侧的两条纪律：默认关，且系统层的几行只在开关里执行。"""

    def test_the_build_flag_defaults_to_off(self):
        text = (REPO / "packaging" / "box" / "build-image.sh").read_text(encoding="utf-8")
        self.assertIn("--preflash", text)
        self.assertIn("PREFLASH=0", text)

    def test_the_service_layer_calls_it_only_under_the_flag(self):
        text = (REPO / "packaging" / "common" / "mrrc-service-layer.sh").read_text(
            encoding="utf-8")
        start = text.index('if [ "${MRRC_PREFLASH:-0}" = "1" ]; then')
        block = text[start:text.index("\nfi", start)]
        self.assertIn("preflash.sh", block,
                      "调用点不在 MRRC_PREFLASH 分支里 ⇒ 默认镜像也会被写入口令")


if __name__ == "__main__":
    unittest.main()
```

创建 `packaging/common/preflash.sh`（写完 `chmod +x`）：

```bash
#!/bin/bash
# 出厂预置（发货版镜像）：固定口令 + 系统层（时区/区域）。
#
# 谁调它：mrrc-service-layer.sh 在 MRRC_PREFLASH=1 时（= `build-image.sh --preflash`）。
# 它在共享服务层里，所以盒子线与树莓派线走同一份实现 —— 但开关默认关，
# 不传 --preflash 的镜像里这个脚本一次都不会被执行。
#
#   bash preflash.sh /opt/mrrc_modern/env/mrrc.env      # 全量（要 root）
#   bash preflash.sh --env-only /tmp/mrrc.env           # 只写 env（单测真跑用，不需要 root）
#
# 幂等：重跑只覆盖同一批键，不动别的任何一行。
set -euo pipefail

# 开关闸门：默认关是"预置"这件事的安全属性，不是可选项 —— 一次误调用就会把
# 公开口令装进一台本来每台口令不同的镜像里。
if [ "${MRRC_PREFLASH:-0}" != "1" ]; then
 echo "preflash: MRRC_PREFLASH is not 1 — refusing to write anything." >&2
 echo "preflash: 这是出厂预置通道；要手动跑请显式 MRRC_PREFLASH=1 …" >&2
 exit 1
fi

ROOT_PASSWORD="${MRRC_PREFLASH_ROOT_PASSWORD:-abcd1234}"
USER_PASSWORD="${MRRC_PREFLASH_USER_PASSWORD:-abcd!1234}"
WEB_PASSWORD="${MRRC_PREFLASH_WEB_PASSWORD:-abcd!1234}"
TIMEZONE="${MRRC_PREFLASH_TIMEZONE:-Asia/Shanghai}"
LOCALE_NAME="${MRRC_PREFLASH_LOCALE:-zh_CN.UTF-8}"
MRRC_USER="${MRRC_USER:-mrrc}"

ENV_ONLY=0
POSITIONAL=()
for arg in "$@"; do
 case "$arg" in
  --env-only) ENV_ONLY=1 ;;
  *) POSITIONAL+=("$arg") ;;
 esac
done
ENV_FILE="${POSITIONAL[0]:-/opt/mrrc_modern/env/mrrc.env}"

log() { printf '\n=== preflash: %s ===\n' "$*"; }

# ── env 文件：Web 口令（唯一的写者）────────────────────────────────────
# 服务层的 sed 已经把 install.sh 在**构建主机**上猜出来的口令剥掉了，所以正常
# 流程里这两个键此刻还不存在。用 python 做"替换或追加"而不是 sed：口令里有
# `!`，而 sed 的替换串对 `&`、`\`、分隔符都敏感 —— 这里没有可被口令破坏的语法。
log "env file: $ENV_FILE"
python3 - "$ENV_FILE" "$WEB_PASSWORD" <<'PY'
import pathlib
import sys

path = pathlib.Path(sys.argv[1])
password = sys.argv[2]
wanted = {"MRRC_WEB_PASSWORD": password, "MRRC_AUTO_PASSWORD": ""}
lines = path.read_text(encoding="utf-8").splitlines() if path.exists() else []
out, seen = [], set()
for line in lines:
    stripped = line.strip()
    if stripped and not stripped.startswith("#") and "=" in line:
        key = line.split("=", 1)[0].strip()
        if key in wanted:
            out.append(f"{key}={wanted[key]}")
            seen.add(key)
            continue
    out.append(line)
for key, value in wanted.items():
    if key not in seen:
        out.append(f"{key}={value}")
path.write_text("\n".join(out) + "\n", encoding="utf-8")
PY

if [ "$ENV_ONLY" = "1" ]; then
 echo "preflash: env-only — 系统层（账号口令/时区/locale）跳过"
 exit 0
fi

# ── 账号口令 ─────────────────────────────────────────────────────────
# 账号由平台壳先建好（盒子的 adduser --system mrrc / Pi 的 FIRST_USER_NAME），
# 服务层的前置断言已经保证它在。不在这里建账号：那是平台的事。
log "accounts: fixed passwords"
id -u "$MRRC_USER" >/dev/null 2>&1 || {
 echo "preflash: no '$MRRC_USER' account — the platform shell must create it" >&2
 exit 1
}
printf 'root:%s\n%s:%s\n' "$ROOT_PASSWORD" "$MRRC_USER" "$USER_PASSWORD" | chpasswd

# ── 时区 ─────────────────────────────────────────────────────────────
# 写文件才是权威：chroot 里没有 D-Bus，timedatectl 是客户端、必然失败
# （与 hostname 同样的道理，见 mrrc-service-layer.sh）。
log "time zone: $TIMEZONE"
[ -f "/usr/share/zoneinfo/$TIMEZONE" ] || {
 echo "preflash: /usr/share/zoneinfo/$TIMEZONE is missing" >&2
 exit 1
}
ln -sf "/usr/share/zoneinfo/$TIMEZONE" /etc/localtime
printf '%s\n' "$TIMEZONE" >/etc/timezone
timedatectl set-timezone "$TIMEZONE" 2>/dev/null || true

# ── 区域/语言 ────────────────────────────────────────────────────────
log "locale: $LOCALE_NAME"
[ -f /etc/locale.gen ] || {
 echo "preflash: /etc/locale.gen is missing (the locales package?)" >&2
 exit 1
}
# 基底可能带一份被裁过的 locale.gen（pi-gen 会裁）：缺就补一行，别失败 ——
# 这条脚本存在的理由是"装出能用的镜像"，不是"验基底的清单"。
if ! grep -q "^${LOCALE_NAME} " /etc/locale.gen; then
 if grep -q "^# *${LOCALE_NAME} " /etc/locale.gen; then
  sed -i "s|^# *${LOCALE_NAME} |${LOCALE_NAME} |" /etc/locale.gen
 else
  echo "preflash: appending ${LOCALE_NAME} to /etc/locale.gen (not present)"
  printf '%s UTF-8\n' "$LOCALE_NAME" >>/etc/locale.gen
 fi
fi
locale-gen
printf 'LANG=%s\n' "$LOCALE_NAME" >/etc/default/locale

# NTP：Armbian 默认已启用 systemd-timesyncd，这里只做显式声明；真判据在机器上
# （`timedatectl` 的 System clock synchronized）而不是构建期。
systemctl enable systemd-timesyncd.service 2>/dev/null || \
 echo "preflash: systemd-timesyncd was not enabled (check the base image)"

echo "preflash: done"
```

```javascript
  // 后台轮询（server 的 _cloud_start_autoconnect，30 s）负责"批准后自己接通"，
  // 所以这里不需要把页面开着。
  function renderCloud(cloud) {
    var grid = document.querySelector('#cloud-grid');
    var pill = document.querySelector('#cloud-pill');
    grid.innerHTML = '';
    Box.setText('#cloud-portal', cloud.portal || '—');
    if (cloud.entry) {
      pill.textContent = '已接通';
      pill.className = 'pill good';
      kv(grid, '呼号', cloud.callsign || '—');
      kv(grid, '入口', cloud.entry);
      if (cloud.cert_reload_required) {
        kv(grid, '提示', '证书刚换 —— 重启盒子后入口才生效', 'bad-text');
      }
    } else if (cloud.has_token) {
      pill.textContent = '等待批准';
      pill.className = 'pill';
      kv(grid, '呼号', cloud.callsign || '—');
      kv(grid, '状态', '已提交，等待运维在门户批准');
    } else {
      pill.textContent = '未申请';
      pill.className = 'pill';
    }
    if (cloud.tunnel_error) kv(grid, '隧道', cloud.tunnel_error, 'bad-text');
    var input = document.querySelector('#cloud-callsign');
    if (input && !input.value && cloud.callsign) input.value = cloud.callsign;
    Box.setText('#cloud-apply', cloud.has_token ? '查询状态' : '申请公网入口');
  }

  function loadCloud() {
    Box.api('/api/cloud/state').then(renderCloud).catch(function (error) {
      if (error.status === 401) { location.href = '/login?next=/settings'; return; }
      Box.message('#cloud-msg', error.message, 'bad');
    });
  }

  Box.on('#cloud-apply', 'click', function () {
    var input = document.querySelector('#cloud-callsign');
    var callsign = (input.value || '').trim().toUpperCase();
    if (!callsign) {
      Box.message('#cloud-msg', '请先填呼号', 'bad');
      return;
    }
    Box.disabled('#cloud-apply', true);
    Box.api('/api/cloud/state').then(function (cloud) {
      // 已经有令牌就不再申请：门户对同一呼号的重复申请是**查重拒绝**的
      // （store.apply），重提只会把一句"已存在申请"糊到用户脸上。
      if (cloud.has_token) return refreshCloud();
      return Box.api('/api/cloud/apply', { method: 'POST', body: { callsign: callsign } })
        .then(function (reply) {
          Box.message('#cloud-msg', '已提交（' + (reply.status || 'applied')
            + '）。等运维在门户批准，盒子会自己接通。', 'good');
          loadCloud();
        });
    }).catch(function (error) {
      Box.message('#cloud-msg', error.message, 'bad');
    }).then(function () { Box.disabled('#cloud-apply', false); });
  });

  function refreshCloud() {
    return Box.api('/api/cloud/refresh', { method: 'POST' }).then(function (state) {
      if (state.connected) {
        Box.message('#cloud-msg', '已接通：' + (state.entry || ''), 'good');
      } else if (state.status === 'rejected') {
        Box.message('#cloud-msg', '门户没有通过这条申请 —— 请联系运维（可以换一个呼号）。', 'bad');
      } else {
        Box.message('#cloud-msg', '还没批准（' + (state.status || 'applied') + '）。', '');
      }
      loadCloud();
    });
  }

  function pollCloud() {
    Box.api('/api/cloud/state').then(function (cloud) {
      renderCloud(cloud);
      if (cloud.has_token && !cloud.entry) setTimeout(pollCloud, 30000);
    }).catch(function () {});
  }
```

```python
def needs_first_run(env: dict[str, str], port_exists=None) -> bool:
    """True when first-launch auto-config should run.

    Returns False (no re-probe) only when the config is genuinely settled:
    a non-default password, a valid radio model, either the port was confirmed
    by a live probe (``MRRC_PORT_CONFIRMED=1``) or it is a real non-default
    device name, **and the device that name points at is on this machine right
    now**. That last clause is what makes a shipped box self-heal: the seller's
    bench and the customer's desk enumerate the FT-710's two CP210x ports in
    their own order, so a recorded name alone is not evidence.

    ``port_exists`` is injectable because this function is otherwise pure and
    the tests must not depend on a /dev node being present on the build host.
    """
    pwd = env.get("MRRC_WEB_PASSWORD", "")
    port = env.get("MRRC_SERIAL_PORT", "").strip()
    model = env.get("MRRC_RADIO_MODEL", "").strip().lower()
    exists = port_exists or port_node_exists
    configured = (
        pwd and pwd != DEFAULT_WEB_PASSWORD
        and model in VALID_MODELS
        and (
            env.get("MRRC_PORT_CONFIRMED") == "1"
            or (port and port not in DEFAULT_SERIAL_PORTS)
        )
        and exists(port)
    )
    return not configured


def port_node_exists(port: str) -> bool:
    """Whether this machine currently has the device the port name points at.

    Only POSIX device paths are ever *claimed absent*. A Windows port name
    ("COM5") is not a filesystem path at all, and "I cannot tell" has to read as
    present here — reading it as absent would make every Windows launch re-probe
    the radio and rewrite the env file.
    """
    if not port:
        return False
    if not port.startswith("/"):
        return True
    return os.path.exists(port)
```

```python
def apply_first_run(
    env: dict[str, str],
    config_path: Path,
    *,
    open_func=serial.Serial,
    preseeded: bool = False,
) -> dict[str, str]:
    """Fill password/serial/model, persist, mark done. Returns updated env.

    ``preseeded`` (MRRC_PRESEED=1) keeps the password decision outside this
    function — the operator or the image already made it. Everything else still
    runs: the probe is the one thing only this machine can do.
    """
    updates: dict[str, str] = {}

    pwd = env.get("MRRC_WEB_PASSWORD", "")
    if not preseeded and (not pwd or pwd == DEFAULT_WEB_PASSWORD):
        pwd = generate_password()
        env["MRRC_WEB_PASSWORD"] = pwd
        updates["MRRC_WEB_PASSWORD"] = pwd
        env["MRRC_AUTO_PASSWORD"] = "1"
        updates["MRRC_AUTO_PASSWORD"] = "1"
```

```python
def main() -> int:
    """Entry point for mrrc-firstboot.service (runs as root).

    ``MRRC_PRESEED=1`` means "this env file is authoritative — do not invent a
    password", and nothing more. It used to return before the probe, and the
    wrapper passes it on every image whose env file exists (i.e. every image
    install.sh built) — so the serial probe never ran on a real box (2026-10-17
    artifact check: env/mrrc.env present, no MRRC_WEB_PASSWORD, no
    MRRC_SERIAL_PORT, no firstboot-done).
    """
    env_path = Path(os.environ.get("MRRC_ENV_FILE", "/opt/mrrc_modern/env/mrrc.env"))
    preseeded = os.environ.get("MRRC_PRESEED") == "1"
    env = dict(
        line.split("=", 1) for line in read_env_text(env_path)[0].splitlines()
        if line.strip() and not line.strip().startswith("#") and "=" in line
    )
    if not needs_first_run(env):
        print("[mrrc-firstboot] config settled; skipping auto-config")
        return 0
```
