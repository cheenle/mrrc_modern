```python
"""NanoVNA 测量结果的纯数学层：没有任何 I/O，没有设备，没有 HTTP。

S11 是复数反射系数 Γ。所有输出都由它派生，所以存盘只存原始 S11（能算出来的
东西不存第二份）。
"""
from __future__ import annotations

import math
from typing import Iterable, Sequence

Z0 = 50.0
SWR_CAP = 99.9          # 开路不是无穷大，是"超出量程"
SUSPECT_NEIGHBOUR_SWR = 1.35   # 孤立 1.00 的掉读判据（antenna-sweep 的既有规则）


def _mag(re: float, im: float) -> float:
    return math.hypot(re, im)


def swr(re: float, im: float) -> float:
    g = _mag(re, im)
    if g >= 1.0:
        return SWR_CAP
    return (1.0 + g) / (1.0 - g)


def to_z(re: float, im: float) -> tuple[float, float]:
    """Γ -> Z = Z0(1+Γ)/(1-Γ)，返回 (R, X)。"""
    den = (1.0 - re) ** 2 + im ** 2
    if den < 1e-18:
        return (float("inf"), 0.0)
    num_re = 1.0 - re * re - im * im
    num_im = 2.0 * im
    return (Z0 * num_re / den, Z0 * num_im / den)


def analyse(freqs: Sequence[float], s11: Sequence[tuple[float, float]]) -> dict:
    """一次扫描的前端载荷。前端不重算任何物理量。"""
    swrs = [swr(re, im) for re, im in s11]
    zs = [to_z(re, im) for re, im in s11]
    return {
        "swr": [round(v, 4) for v in swrs],
        "z": [[None if math.isinf(r) else round(r, 3), round(x, 3)] for r, x in zs],
        "resonances": resonances(freqs, swrs),
        "suspect": suspect_points(swrs),
    }


def resonances(freqs: Sequence[float], swrs: Sequence[float],
               floor: float = 1.2, ceiling: float = 2.0) -> list[dict]:
    """连续低于 floor 的游程 = 一个谐振平台，取游程中心频率与最低点。

    孤立一个点不算（它通常是读数事故，见 suspect_points）。
    """
    out: list[dict] = []
    i, n = 0, len(swrs)
    while i < n:
        if swrs[i] > floor:
            i += 1
            continue
        j = i
        while j + 1 < n and swrs[j + 1] <= floor:
            j += 1
        if j - i >= 1:
            lo = min(range(i, j + 1), key=lambda k: swrs[k])
            if swrs[lo] < ceiling:
                out.append({
                    "f_hz": round((freqs[i] + freqs[j]) / 2.0, 1),
                    "f_min_hz": freqs[lo],
                    "swr_min": round(swrs[lo], 3),
                    "width_hz": round(freqs[j] - freqs[i], 1),
                })
        i = j + 1
    return out


def suspect_points(swrs: Sequence[float]) -> list[int]:
    """单点 ≈1.00 而两侧都 > 1.35 —— 掉读，不是谐振。"""
    out: list[int] = []
    for k in range(1, len(swrs) - 1):
        if swrs[k] <= 1.05 and swrs[k - 1] > SUSPECT_NEIGHBOUR_SWR \
                and swrs[k + 1] > SUSPECT_NEIGHBOUR_SWR:
            out.append(k)
    return out


def to_csv(freqs: Sequence[float], s11: Sequence[tuple[float, float]]) -> str:
    lines = ["freq_hz,s11_re,s11_im,swr,r_ohm,x_ohm"]
    for f, (re, im) in zip(freqs, s11):
        r, x = to_z(re, im)
        lines.append("%.6f,%.6f,%.6f,%.4f,%.3f,%.3f"
                     % (f, re, im, swr(re, im), r, x))
    return "\n".join(lines) + "\n"


def to_s1p(freqs: Sequence[float], s11: Sequence[tuple[float, float]]) -> str:
    lines = ["! NanoVNA sweep exported by MRRC Modern",
             "# Hz S RI R 50"]
    lines += ["%.6f %.6f %.6f" % (f, re, im) for f, (re, im) in zip(freqs, s11)]
    return "\n".join(lines) + "\n"


def compare(a: dict, b: dict) -> dict:
    """两次扫描按频率对齐（较粗的一侧线性插值），返回 ΔSWR 与最大差值点。"""
    fa, sa = a["freqs"], a["swr"]
    fb, sb = b["freqs"], b["swr"]
    if len(fa) != len(fb) or fa[0] != fb[0] or fa[-1] != fb[-1]:
        if len(fa) >= len(fb):
            fa, sa, sb = fa, sa, _resample(fb, sb, fa)
        else:
            fa, sa, sb = fb, _resample(fa, sa, fb), sb
    deltas = [y - x for x, y in zip(sa, sb)]
    k = max(range(len(deltas)), key=lambda i: abs(deltas[i]))
    return {"freqs": fa, "delta_swr": [round(d, 4) for d in deltas],
            "worst_hz": fa[k], "worst_delta": round(deltas[k], 4)}


def _resample(freqs: Sequence[float], values: Sequence[float],
              target: Sequence[float]) -> list[float]:
    out: list[float] = []
    j = 0
    for f in target:
        while j + 2 < len(freqs) and freqs[j + 1] < f:
            j += 1
        f0, f1 = freqs[j], freqs[min(j + 1, len(freqs) - 1)]
        v0, v1 = values[j], values[min(j + 1, len(values) - 1)]
        out.append(v0 if f1 == f0 else v0 + (v1 - v0) * (f - f0) / (f1 - f0))
    return out
```

def _frame(points, mask=0x83):
    """按固件 main.c 的布局造一帧二进制 scan_bin 回复。"""
    out = struct.pack("<HH", mask, points)

"""NanoVNA-H 的串口协议（唯一碰串口的模块）。

契约来源：hugen79/NanoVNA-H 的 `main.c`（`cmd_scan` / `cmd_data`），并与真机
1.2.14 双向核对（见 spec §1.2）。-- 二进制帧布局：
    u16 mask, u16 points, 然后每点 u32 freq + 每个启用数组各 f32 re/im
帧长必须精确匹配，否则整帧丢弃（半截数据会画出一条看起来正常的假曲线）。
"""
from __future__ import annotations

import logging
import os
import re
import struct
from dataclasses import dataclass, field
from pathlib import Path
from typing import Optional, Sequence

log = logging.getLogger(__name__)

MASK_FREQ = 0x01
MASK_S11 = 0x02
MASK_S21 = 0x04
MASK_BINARY = 0x80
POINTS_ALLOWED = (101, 201, 401)
FREQ_MIN_HZ = 50_000
FREQ_MAX_HZ = 900_000_000
NANO_VID_PID = "0483:5740"
DFU_VID_PID = "0483:df11"


class FrameError(Exception):
    """整帧不完整或多余 —— 一律丢弃重扫。"""


@dataclass
class DeviceInfo:
    board: str = ""
    version: str = ""
    build: str = ""
    platform: str = ""
    points_max: int = 0


@dataclass
class Sweep:
    freqs: list                 # list[int]
    s11: list                   # list[tuple[float, float]]
    s21: Optional[list] = None
    elapsed_ms: int = 0


def scan_command(start: int, stop: int, points: int, s21: bool = False) -> str:
    if not (FREQ_MIN_HZ <= start < stop <= FREQ_MAX_HZ):
        raise ValueError("start/stop out of range: %r %r" % (start, stop))
    if points not in POINTS_ALLOWED:
        raise ValueError("points must be one of %r" % (POINTS_ALLOWED,))
    mask = MASK_BINARY | MASK_FREQ | MASK_S11 | (MASK_S21 if s21 else 0)
    return "scan_bin %d %d %d %d" % (start, stop, points, mask)

# ── NanoVNA-H 天线诊断台（spec 2026-10-18）────────────────────────────
# 一台串口仪器：不是电台，不共享任何 CAT 状态。串口只在 nano/vna.py 里出现；
# 本文件负责四件事 —— 一次一件、发射中闭嘴、不抢 CAT 口、角色隔离。
NANO_DIR = Path(_env("MRRC_NANO_DIR", str(_writable_runtime_dir() / "nano")))
NANO_PORT = _env("MRRC_NANO_PORT", "")
NANO_ENABLED = (_env("MRRC_NANO", "on") or "on").strip().lower() not in ("off", "0", "false", "no")

_nano_device: Optional[NanoVna] = None
_nano_lock = asyncio.Lock()          # 一次一件
_nano_last_sweep: Optional[dict] = None


def _nano_tx_active() -> bool:
    """电台在发射（PTT / TUNE / FT8 / 一键 CQ）时，绝不能碰接在天线上的 VNA。"""
    if radio is not None and radio.tx_status:
        return True
    try:
        if _ft8_transmitting():
            return True
    except Exception:                                              # noqa: BLE001
        pass
    try:
        if _cq_player.is_calling:
            return True
    except Exception:                                              # noqa: BLE001
        pass
    return False


def _nano_require_operator(request: Request) -> Optional[JSONResponse]:
    """页面与接口共用的前置门：主站会话，允许 listen/sdr 之外的一切。"""
    if not NANO_ENABLED:
        return JSONResponse({"error": "nano disabled"}, status_code=404)
    if _is_listen_request(request) or _sdr_token_from_request(request) in _sdr_tokens:
        return JSONResponse({"error": "forbidden"}, status_code=403)
    return None
```

@app.post("/api/nano/scan")
async def api_nano_scan(request: Request):
    gate = _nano_require_operator(request)
    if gate is not None:
        return gate
    if _nano_tx_active():
        return JSONResponse({"error": "tx_active",
                             "detail": "电台在发射，已锁定扫描"}, status_code=409)
    if _nano_lock.locked():
        return JSONResponse({"error": "busy",
                             "detail": "另一次操作正在进行"}, status_code=409)
    body = await request.json()
    try:
        start = int(body["start"]); stop = int(body["stop"])
        points = int(body.get("points", 401))
        cmd = scan_command(start, stop, points)
    except (KeyError, TypeError, ValueError) as exc:
        return JSONResponse({"error": "bad_request", "detail": str(exc)}, status_code=400)
    async with _nano_lock:
        device = await asyncio.to_thread(_nano_ensure_device)
        sweep = await asyncio.to_thread(device.scan, start, stop, points)
        await asyncio.to_thread(device.restore_sweep)
    payload = {"freqs": sweep.freqs, "s11": sweep.s11,
               "elapsed_ms": sweep.elapsed_ms, **analyse(sweep.freqs, sweep.s11)}
    global _nano_last_sweep
    _nano_last_sweep = {"freqs": sweep.freqs, "s11": sweep.s11,
                        "start_hz": start, "stop_hz": stop, "points": points}
    return JSONResponse(payload)
```
