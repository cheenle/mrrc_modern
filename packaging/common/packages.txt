# System packages every MRRC Modern image needs. One answer to "what does the
# server need", installed by packaging/common/mrrc-service-layer.sh on both lines.
#
# Measured, not assumed: on the pinned Armbian base (packaging/box/, 2026-10-11)
# install.sh could not create its virtualenv because python3.11-venv, pip, the
# PortAudio/ALSA headers and libopus0 were all absent — only libasound2, gcc,
# make, rsync, git, curl and sudo were there. install.sh cannot repair that
# itself: it creates the virtualenv in STEP 2 and installs system packages in
# STEP 3, so a missing ensurepip is fatal before apt ever runs, and a venv made
# without it has no pip at all.
#
# libopus0 is the one that is easy to miss and expensive to miss: without it the
# server silently falls back to PCM instead of Opus.
#
# python3.11-venv is deliberately pinned to the base's Python (both bases are
# bookworm / 3.11 today). A base bump must re-measure this file; apt fails loudly
# on a package name that no longer exists, which is the outcome we want.
python3.11-venv
python3-dev
portaudio19-dev
libportaudio2
libasound2-dev
libopus0
libopus-dev
avahi-daemon

# logrotate: server-stdout.log is written by systemd's `append:` target and has
# no size bound of its own (a unit cannot express one). The daemon runs on the
# mrrc-logrotate.timer that mrrc-service-layer.sh installs; without this package
# that timer fails and the file grows until the disk is full.
logrotate

# ── FT8 work mode: the native DSP library ────────────────────────────────
# The FT8 engine loads libwsjt_core.so through ctypes, and the images build it
# on the target during the build (packaging/common/mrrc-service-layer.sh) so a
# prebuilt binary never has to be kept in step with the sources by hand.  This
# toolchain is what a build needs; it stays installed because
# linux/mrrc_update.sh re-runs ft8/dsp/build.sh after every update — its source
# digest makes that a no-op unless ft8/dsp actually moved, and it is the only
# thing that keeps an updated tree from pairing new Python with an old .so.
#
# gfortran pulls libgfortran5/libgomp1, libfftw3-dev pulls libfftw3-single3:
# those are the runtime side, and they must not be purged as "build only".
#
# No libboost-dev: the two Boost headers the DSP needs are vendored under
# ft8/dsp/vendor-boost/ (tools/ft8/vendor_boost_headers.sh).  Debian only ships
# <boost/crc.hpp> inside libboost1.74-dev — 138 MB of headers, in both images,
# for two files.
gfortran
g++
cmake
pkg-config
libfftw3-dev
