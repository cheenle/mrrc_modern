# ── Web Server ────────────────────────────────────────────────────────────
fastapi>=0.100.0
uvicorn[standard]>=0.23.0
websockets>=12.0

# ── Serial CAT Protocol ───────────────────────────────────────────────────
# Requires Silicon Labs CP210x USB-UART driver on the host OS.
# macOS: built-in (AppleUSBCDC); Linux: built-in (cp210x kernel module);
# Windows: download from silabs.com/developers/usb-to-uart-bridge-vcp-drivers
pyserial>=3.5

# ── Audio I/O (RX capture + TX playback, 44.1 kHz mono) ───────────────────
# Requires the PortAudio C library on the host OS:
#   macOS:  brew install portaudio
#   Linux:  apt install portaudio19-dev  (or portaudio-devel on Fedora)
#   Windows: bundled with PyAudio wheel (no separate install)
pyaudio>=0.2.11

# ── FT8 mode (mrrc_modern × MRRC-FT8 engine, feat/ft8) ─────────────────────
# scipy: DSP-side signal processing ported with the engine (FIR/filters).
# soxr:  streaming resampler for the 44.1 kHz -> 12 kHz RX tap.
#
# Upper bound is load-bearing (2026-10-26): scipy 1.17.1's aarch64 wheel dies with
# SIGSEGV on the W103D's Cortex-A53 the moment `import scipy.signal` runs, so the
# service sits in a restart loop and the web UI never answers
# (`Main process exited, code=killed, status=11/SEGV`, 97 restarts, measured on a
# freshly flashed 1.28.6). That CPU advertises `fp asimd evtstrm aes pmull sha1 sha2
# crc32 cpuid` -- plain ARMv8.0, no dot-product/fp16. 1.14.1 verified good on the
# same box; the faulthandler trace pointed at ft8/engine/audio_rx.py:21.
scipy>=1.12,<1.15
soxr>=0.4

# ── Numeric Processing ─────────────────────────────────────────────────────
# Used for: audio resampling (44.1k ↔ 48k linear interpolation),
#           S-meter-based synthetic spectrum fallback (Gaussian peaks).
numpy>=1.24.0

# ── Recording (server-side QSO MP3) ─────────────────────────────────────
# Bundles libmp3lame as a prebuilt wheel — no ffmpeg dependency.  Wheels
# exist for macOS arm64/x86_64, Windows x64 and Linux aarch64 (the
# Raspberry Pi image).
lameenc>=1.8.0

# ── TLS/SSL (HTTPS) ─────────────────────────────────────────────────────
# Used by ssl_bootstrap.py: the desktop launcher generates a self-signed
# certificate on first run so the app starts on HTTPS by default.
cryptography>=41.0.0

# ── Remote help (Call help) ──────────────────────────────────────────────
# The customer-initiated SSH door (help_session.py). Parameterized because the
# four platforms must behave the same: Windows has no sshd to fall back on, so
# the app carries the server itself. cryptography (above) does the crypto;
# this only adds the SSH transport and channel layer.
paramiko>=3.4.0
